Create a function incrementFollowers(int *followers, int n) that increases each follower count in an array (representing Instagram followers for 5 friends) by 100 using pointer arithmetic, then print the updated counts. #include void incrementFollowers(int *followers, int n) { for (int i = 0; i < n; i++) { *followers = *followers + 100; followers++; } } int main() { int followers[5] = {1000, 2500, 1800, 3200, 1500}; incrementFollowers(followers, 5); printf("Updated follower counts:\n"); for (int i = 0; i < 5; i++) { printf("%d\n", followers[i]); } return 0; }
