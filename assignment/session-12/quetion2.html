Create a structure called FoodItem to store Zomato-style menu data: itemName (string), price (float), and rating (float). Initialize an array of 3 FoodItem variables with real menu items and display their details using a loop. #include struct FoodItem { char itemName[50]; float price; float rating; }; int main() { struct FoodItem menu[3] = { {"Masala Dosa", 120.50, 4.5}, {"Paneer Tikka", 250.00, 4.3}, {"Veg Biryani", 180.75, 4.6} }; for (int i = 0; i < 3; i++) { printf("Item: %s\n", menu[i].itemName); printf("Price: ₹%.2f\n", menu[i].price); printf("Rating: %.1f\n", menu[i].rating); printf("-------------------\n"); } return 0; }
